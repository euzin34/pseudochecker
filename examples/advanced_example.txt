// Advanced CIE 9618 Pseudocode Examples
// Demonstrating arrays, procedures, functions, and advanced features

// Constants
CONSTANT PI = 3.14159
CONSTANT MAX_SIZE = 100

// Variable declarations
DECLARE x : INTEGER
DECLARE y : REAL
DECLARE name : STRING
DECLARE initial : CHAR
DECLARE isValid : BOOLEAN

// Array declarations
DECLARE numbers : ARRAY[1:10] OF INTEGER
DECLARE matrix : ARRAY[1:5, 1:5] OF REAL
DECLARE names : ARRAY[0:9] OF STRING

// Initialize variables
x <- 10
y <- 3.14
name <- "Bob"
initial <- 'B'
isValid <- TRUE

// Array operations
numbers[1] <- 42
numbers[2] <- 84
matrix[1, 1] <- 1.5
names[0] <- "First"

// Complex conditional
IF isValid AND x > 0 THEN
  OUTPUT "Valid and positive"
ENDIF

// CASE statement
CASE OF x
  1 : OUTPUT "One"
  2 : OUTPUT "Two"
  10 : OUTPUT "Ten"
ENDCASE

// FOR loop with STEP
FOR i <- 1 TO 10 STEP 2
  OUTPUT i
NEXT i

// Negative STEP
FOR j <- 10 TO 1 STEP -1
  numbers[j] <- j * 2
NEXT j

// REPEAT-UNTIL loop
REPEAT
  OUTPUT "Enter a positive number: "
  INPUT x
UNTIL x > 0

// Expressions with operators
x <- (5 + 3) * 2
y <- x / 2.0
x <- x MOD 3
x <- x DIV 2

// Boolean expressions
isValid <- x > 5 AND y < 10
isValid <- NOT isValid OR x = 0

// Array access in expressions
x <- numbers[1] + numbers[2]
OUTPUT numbers[1]
